init_trampolines:
    LEA     ROM_Source_Table(PC), A0    ; Onde estão os dados/instruções na ROM
    MOVEA.L #0x0007D400, A1             ; Endereço base onde os JMPs vão morar na RAM
    MOVEQ   #5, D0                      ; Quantidade de vetores para copiar (ex: 6 vetores)

copy_loop:
    ; Como o JMP absoluto ocupa 6 bytes, copiamos palavra por palavra ou long + word
    MOVE.L  (A0)+, (A1)+                ; Copia a instrução JMP (opc) + parte do endereço
    MOVE.W  (A0)+, (A1)+                ; Copia o resto do endereço absoluto
    DBRA    D0, copy_loop
    RTS

; Tabela pré-formatada na ROM com os opcodes do JMP (4EF9) seguidos dos endereços dos handlers
ROM_Source_Table:
    ; Vetor 1 (Bus Error) -> JMP 0x007E0000 (exemplo de endereço do handler)
    DC.W    0x4EF9                      ; Opcode para JMP absoluto longo
    DC.L    0x007E0000                  ; Endereço real do handler na RAM

    ; Vetor 2 (Address Error) -> JMP 0x007E0006
    DC.W    0x4EF9
    DC.L    0x007E0006

    ; Vetor 3 (Illegal Instruction) -> JMP 0x007E000C
    DC.W    0x4EF9
    DC.L    0x007E000C
    ; ... repita para os demais vetores
